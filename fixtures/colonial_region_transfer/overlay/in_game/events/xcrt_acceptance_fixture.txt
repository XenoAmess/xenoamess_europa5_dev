namespace = xcrt_acceptance

xcrt_acceptance.1 = {
	type = country_event
	title = xcrt_acceptance.1.title
	desc = xcrt_acceptance.1.desc
	outcome = neutral
	orphan = yes
	trigger = { always = no }

	option = {
		name = xcrt_acceptance.1.a

		location:tortuga = {
			discover_location = root
			create_country_from_location = {
				overlord = root
				subject_type = subject_type:colonial_nation
				define_unique_country_tag = XCRTT
				change_country_name = XCRTT
				change_country_adjective = XCRTT
			}
		}

		location:guahaba = {
			discover_location = root
			create_country_from_location = {
				overlord = root
				subject_type = subject_type:vassal
				define_unique_country_tag = XCRTD
				change_country_name = XCRTD
				change_country_adjective = XCRTD
				set_global_variable = {
					name = xcrt_acceptance_direct_donor
					value = this
				}
			}
		}

		location:porto_santo = {
			discover_location = root
			change_location_owner = global_var:xcrt_acceptance_direct_donor
		}

		location:baynoa = {
			discover_location = root
			create_country_from_location = {
				overlord = global_var:xcrt_acceptance_direct_donor
				subject_type = subject_type:vassal
				define_unique_country_tag = XCRTS
				change_country_name = XCRTS
				change_country_adjective = XCRTS
			}
		}

		location:iguamuco = {
			discover_location = root
			change_location_owner = root
			create_country_from_location = {
				define_unique_country_tag = XCRTI
				change_country_name = XCRTI
				change_country_adjective = XCRTI
			}
		}

		location:marien = {
			discover_location = root
			change_location_owner = root
		}

		set_global_variable = {
			name = xcrt_acceptance_actor
			value = root
		}
		set_global_variable = xcrt_acceptance_ready
	}
}

xcrt_acceptance.2 = {
	type = country_event
	title = xcrt_acceptance.2.title
	desc = xcrt_acceptance.2.desc
	outcome = neutral
	orphan = yes
	trigger = { always = no }

	option = {
		name = xcrt_acceptance.2.pass
		trigger = {
			has_global_variable = xcrt_acceptance_ready
			location:tortuga = { owner = c:XCRTT }
			location:marien = { owner = c:XCRTT }
			location:guahaba = { owner = c:XCRTT }
			location:baynoa = { owner = c:XCRTT }
			location:iguamuco = { owner = c:XCRTI }
			location:porto_santo = { NOT = { owner ?= c:XCRTT } }
			country_exists = c:XCRTT
			NOT = { country_exists = c:XCRTS }
			country_exists = c:XCRTI
			c:XCRTT = { is_subject_of = global_var:xcrt_acceptance_actor }
		}
		custom_tooltip = xcrt_acceptance.audit.pass.tt
	}

	option = {
		name = xcrt_acceptance.2.fail
		trigger = {
			NAND = {
				has_global_variable = xcrt_acceptance_ready
				location:tortuga = { owner = c:XCRTT }
				location:marien = { owner = c:XCRTT }
				location:guahaba = { owner = c:XCRTT }
				location:baynoa = { owner = c:XCRTT }
				location:iguamuco = { owner = c:XCRTI }
				location:porto_santo = { NOT = { owner ?= c:XCRTT } }
				country_exists = c:XCRTT
				NOT = { country_exists = c:XCRTS }
				country_exists = c:XCRTI
				c:XCRTT = { is_subject_of = global_var:xcrt_acceptance_actor }
			}
		}
		custom_tooltip = xcrt_acceptance.audit.fail.tt
	}
}
