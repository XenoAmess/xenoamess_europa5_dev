# Fresh-game scenario: the target is a normal vassal rather than a colony.
xcrt_acceptance.8 = {
	type = country_event
	title = xcrt_acceptance.8.title
	desc = xcrt_acceptance.8.desc
	outcome = neutral
	orphan = yes
	trigger = { always = no }

	option = {
		name = xcrt_acceptance.8.a

		location:tortuga = {
			discover_location = root
			create_country_from_location = {
				overlord = root
				subject_type = subject_type:vassal
				define_unique_country_tag = XCRTT
				change_country_name = XCRTT
				change_country_adjective = XCRTT
			}
		}
		location:guahaba = {
			discover_location = root
			create_country_from_location = {
				overlord = root
				subject_type = subject_type:vassal
				define_unique_country_tag = XCRTD
				change_country_name = XCRTD
				change_country_adjective = XCRTD
				set_global_variable = {
					name = xcrt_acceptance_direct_donor
					value = this
				}
			}
		}
		location:porto_santo = {
			discover_location = root
			change_location_owner = global_var:xcrt_acceptance_direct_donor
		}
		location:baynoa = {
			discover_location = root
			create_country_from_location = {
				overlord = global_var:xcrt_acceptance_direct_donor
				subject_type = subject_type:vassal
				define_unique_country_tag = XCRTS
				change_country_name = XCRTS
				change_country_adjective = XCRTS
			}
		}
		location:iguamuco = {
			discover_location = root
			change_location_owner = root
			create_country_from_location = {
				define_unique_country_tag = XCRTI
				change_country_name = XCRTI
				change_country_adjective = XCRTI
			}
		}
		location:marien = {
			discover_location = root
			change_location_owner = root
		}
		set_global_variable = {
			name = xcrt_acceptance_actor
			value = root
		}
		set_global_variable = xcrt_acceptance_phase2_vassal_ready
	}
}

# Fresh-game Tusi scenario. LNG keeps its historical direct GYT Tusi relation,
# temporarily moves GYT's capital to the isolated Caribbean Region, and brings
# it to 14 locations while two transferable locations remain in that Region.
xcrt_acceptance.10 = {
	type = country_event
	title = xcrt_acceptance.10.title
	desc = xcrt_acceptance.10.desc
	outcome = neutral
	orphan = yes
	trigger = { always = no }

	option = {
		name = xcrt_acceptance.10.a
		trigger = {
			tag = LNG
			c:GYT = {
				is_subject_of = root
				is_subject_type = tusi
				num_locations = 3
			}
		}

		location:tortuga = { discover_location = root change_location_owner = c:GYT }
		c:GYT = { set_capital = location:tortuga }
		location:iguamuco = {
			discover_location = root
			change_location_owner = root
			create_country_from_location = {
				define_unique_country_tag = XCRTI
				change_country_name = XCRTI
				change_country_adjective = XCRTI
			}
		}

		location:torres_vedras = { discover_location = root change_location_owner = c:GYT }
		location:santarem = { discover_location = root change_location_owner = c:GYT }
		location:evora = { discover_location = root change_location_owner = c:GYT }
		location:setubal = { discover_location = root change_location_owner = c:GYT }
		location:alcacer_do_sal = { discover_location = root change_location_owner = c:GYT }
		location:beja = { discover_location = root change_location_owner = c:GYT }
		location:sines = { discover_location = root change_location_owner = c:GYT }
		location:odemira = { discover_location = root change_location_owner = c:GYT }
		location:ourique = { discover_location = root change_location_owner = c:GYT }
		location:elvas = { discover_location = root change_location_owner = c:GYT }
		location:marien = { discover_location = root change_location_owner = root }
		location:guahaba = { discover_location = root change_location_owner = root }

		set_global_variable = {
			name = xcrt_acceptance_actor
			value = root
		}
		set_global_variable = xcrt_acceptance_phase2_tusi_ready
	}
}
