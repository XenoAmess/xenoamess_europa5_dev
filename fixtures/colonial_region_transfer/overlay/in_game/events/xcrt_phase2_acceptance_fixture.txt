# Turn the 14 -> 16 disabled setup into a 14 -> 15 success setup. Keep GYT's
# native relation and Nixi capital, move LNG's capital out of South China, and
# isolate other LNG subject-tree donors in this Region from the one test donor.
xcrt_acceptance.11 = {
	type = country_event
	title = xcrt_acceptance.11.title
	desc = xcrt_acceptance.11.desc
	outcome = neutral
	orphan = yes
	trigger = { always = no }

	option = {
		name = xcrt_acceptance.11.a
		trigger = {
			has_global_variable = xcrt_acceptance_phase2_tusi_ready
			c:GYT = {
				is_subject_type = tusi
				num_locations = 14
			}
		}
		location:linxi = {
			change_location_owner = c:XCRTI
		}
		set_capital = location:porto_santo
		c:GYT = {
			capital.region = {
				every_location_in_region = {
					limit = {
						is_ownable = yes
						NOT = { owner ?= c:GYT }
						NOT = { this = location:tongan_lijiang }
						owner ?= {
							OR = {
								this = root
								is_subject_or_below_of = root
							}
						}
					}
					add_to_list = xcrt_acceptance_extra_donors_to_isolate
				}
			}
		}
		every_in_list = {
			list = xcrt_acceptance_extra_donors_to_isolate
			change_location_owner = c:XCRTI
		}
	}
}
