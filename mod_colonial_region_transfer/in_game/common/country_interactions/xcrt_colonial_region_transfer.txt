xcrt_cleanup_colonial_region = {
	type = subject
	category = CATEGORY_SUBJECT_ACTIONS

	ai_tick = never
	use_enroute = no
	block_when_at_war = yes
	show_message = no
	show_message_to_target = no

	potential = {
		scope:actor = {
			any_subject = {
				xcrt_is_eligible_direct_subject_target = yes
			}
		}
	}

	allow = {
		scope:actor = {
			at_war = no
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_subject = {
					limit = {
						xcrt_is_eligible_direct_subject_target = yes
					}
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "xcrt_select_subject"
		none_available_msg_key = "xcrt_no_subject"
		show_why_not_enabled = yes
		column = {
			data = name
		}
		column = {
			data = population
		}
		visible = {
			xcrt_is_eligible_direct_subject_target = yes
		}
		enabled = {
			custom_tooltip = {
				text = xcrt_target_must_be_at_peace_tt
				at_war = no
			}
			custom_tooltip = {
				text = xcrt_overlord_capital_outside_region_tt
				scope:actor = {
					capital = {
						NOT = { region = scope:recipient.capital.region }
					}
				}
			}
			custom_tooltip = {
				text = xcrt_affected_subjects_at_peace_tt
				scope:actor = {
					NOT = {
						any_subject_or_below = {
							at_war = yes
							any_owned_location = {
								region = scope:recipient.capital.region
							}
						}
					}
				}
			}
			custom_tooltip = {
				text = xcrt_has_transferable_locations_tt
				scope:recipient.capital.region = {
					any_location_in_region = {
						xcrt_is_transferable_location = yes
					}
				}
			}
			custom_tooltip = {
				text = xcrt_subject_location_limit_tt
				xcrt_recipient_respects_tusi_cap = yes
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:recipient
				scope:recipient = {
					xcrt_is_eligible_direct_subject_target = yes
				}
			}
			scope:recipient.capital.region = {
				every_location_in_region = {
					limit = {
						xcrt_is_transferable_location = yes
					}
					add_to_list = xcrt_transfer_locations
				}
			}
			if = {
				limit = {
					scope:actor = {
						at_war = no
						capital = {
							NOT = { region = scope:recipient.capital.region }
						}
						NOT = {
							any_subject_or_below = {
								at_war = yes
								any_owned_location = {
									region = scope:recipient.capital.region
								}
							}
						}
					}
					scope:recipient = {
						at_war = no
						xcrt_is_eligible_direct_subject_target = yes
						xcrt_frozen_transfer_list_respects_tusi_cap = yes
					}
					list_size = { name = xcrt_transfer_locations value > 0 }
				}
				custom_tooltip = {
					text = xcrt_transfer_effect_tt
					every_in_list = {
						list = xcrt_transfer_locations
						change_location_owner = scope:recipient
					}
				}
			}
			else = {
				custom_tooltip = xcrt_requirements_changed_tt
			}
		}
		else = {
			custom_tooltip = xcrt_select_subject_first_tt
		}
	}
}
